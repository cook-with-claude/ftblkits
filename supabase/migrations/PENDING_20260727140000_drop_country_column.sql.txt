-- The trigger exists only to keep the two columns in step; with one column gone
-- it has nothing to do. Drop it before the column so it is never briefly
-- referencing a column that no longer exists.
drop trigger if exists products_sync_team_country on public.products;
drop function if exists public.products_sync_team_country();

alter table public.products drop column if exists country;

-- The old CHECK constraint went with the column, if it had not already been
-- renamed onto `team` by the expand migration.
do $$
begin
  if exists (select 1 from pg_constraint where conname = 'products_country_nonblank') then
    alter table public.products drop constraint products_country_nonblank;
  end if;
end
$$;
